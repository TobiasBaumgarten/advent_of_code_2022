6/// parses the input to coordinate tuples per line
7fn parse_input(input: &str) -> Vec<Vec<(i32, i32)>> {
8    input
9        .lines()
10        .map(|line| {
11            line.replace(" ", "") // replace all whitespaces
12                .split("->") // split the coordinates
13                .map(|s| parse_coord_string(s)) // convert to tuple
14                .collect()
15        })
16        .collect()
17}
18
19/// parses a String with coordinates like ("123,4") to a truple ((123,4))
20fn parse_coord_string(coord_str: &str) -> (i32, i32) {
21    let splitted: Vec<&str> = coord_str.split(",").collect();
22
23    if splitted.len() != 2 {
24        panic!("The coordinates have to be exactly an x and y coordinate separated with a ','");
25    }
26    let x = splitted[0]
27        .parse::<i32>()
28        .expect("x cannot be converted to a int");
29    let y = splitted[1]
30        .parse::<i32>()
31        .expect("y cannot be converted to a int");
32
33    (x, y)
34}
35#[derive(Debug)]
36struct Map {
37    solid_blocks: HashSet<(i32, i32)>,
38    max_y: i32,
39}
40
41impl FromStr for Map {
42    type Err = &'static str;
43
44    fn from_str(s: &str) -> Result<Self, Self::Err> {
45        let paths = parse_input(s);
46        let mut solid_blocks: HashSet<(i32, i32)> = HashSet::new();
47
48        for path in paths {
49            for window in path.windows(2) {
50                let (a, b) = (window[0], window[1]);
51
52                // Create a range that works regardless of direction
53                let (start, end, is_horizontal) = if a.0 == b.0 {
54                    // Vertical line (x values equal)
55                    let min_y = a.1.min(b.1);
56                    let max_y = a.1.max(b.1);
57                    (min_y, max_y, false)
58                } else {
59                    // Horizontal line (y values equal)
60                    let min_x = a.0.min(b.0);
61                    let max_x = a.0.max(b.0);
62                    (min_x, max_x, true)
63                };
64
65                // Inclusive range
66                for i in start..=end {
67                    if is_horizontal {
68                        solid_blocks.insert((i, a.1));
69                    } else {
70                        solid_blocks.insert((a.0, i));
71                    }
72                }
73            }
74        }
75
76        let max_y = solid_blocks.iter().map(|(_, y)| *y).max().unwrap_or(0); // Provide a default in case collection is empty
77
78        Ok(Self {
79            solid_blocks,
80            max_y,
81        })
82    }
83}

85enum FallingResult {
86    Falling((i32, i32)),
87    Rest,
88    Abyss,
89}
90
91impl Map {
92    #[inline]
93    fn is_empty_slot(&self, coord: (i32, i32)) -> bool {
94        !self.solid_blocks.contains(&coord)
95    }
96
97    fn add_solid(&mut self, coord: (i32, i32)) {
98        self.solid_blocks.insert(coord);
99    }
100
101    /// tries to find the next falling position.
102    /// if it doesn't find the next position it saves the block in the solid list
103    fn next(&mut self, current: (i32, i32)) -> FallingResult {
104        match current {
105            (_, y) if self.max_y < y => FallingResult::Abyss,
106            (x, y) if self.is_empty_slot((x, y + 1)) => FallingResult::Falling((x, y + 1)),
107            (x, y) if self.is_empty_slot((x - 1, y + 1)) => FallingResult::Falling((x - 1, y + 1)),
108            (x, y) if self.is_empty_slot((x + 1, y + 1)) => FallingResult::Falling((x + 1, y + 1)),
109            _ => {
110                self.add_solid(current);
111                FallingResult::Rest
112            }
113        }
114    }
115}

117pub fn solve_star_one(input: &str) -> usize {
118    let mut solid_map = Map::from_str(&input).expect("Coulnd parse Map");
119    let source: (i32, i32) = (500, 0);
120    let mut count_sand = 0;
121    let mut curr_sand = source;
122
123    loop {
124        match solid_map.next(curr_sand) {
125            FallingResult::Abyss => break,
126            FallingResult::Falling(curr) => curr_sand = curr,
127            FallingResult::Rest => {
128                count_sand += 1;
129                curr_sand = source;
130            }
131        }
132    }
133    count_sand
134}

136pub fn solve_star_two(input: &str) -> usize {
137    let mut solid_map = Map::from_str(&input).expect("Coulnd parse Map");
138    solid_map.max_y += 1; // changed the max y value
139    let source: (i32, i32) = (500, 0);
140    let mut count_sand = 0;
141    let mut curr_sand = source;
142
143    loop {
144        match solid_map.next(curr_sand) {
145            FallingResult::Abyss => {
146                solid_map.add_solid(curr_sand);
147                curr_sand = source;
148            }
149            FallingResult::Falling(curr) => curr_sand = curr,
150            FallingResult::Rest => {
151                count_sand += 1;
152                if curr_sand == source {
153                    return count_sand;
154                }
155                curr_sand = source;
156            }
157        }
158    }
159}
