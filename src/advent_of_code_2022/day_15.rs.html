5const FREQUENZY_FACTOR: i64 = 4000000;
6
7pub fn solve_puzzle_one(input: impl Into<String>, y: i32) -> i32 {
8    let sensors = parse_input(input);
9    let max_x = sensors
10        .iter()
11        .map(|s| s.position.0 + s.distance() as i32)
12        .max()
13        .unwrap();
14    let min_x = sensors
15        .iter()
16        .map(|s| s.position.0 - s.distance() as i32)
17        .min()
18        .unwrap();
19    count_none_positions(&sensors, y, min_x, max_x)
20}

22pub fn solve_puzzle_two(input: impl Into<String>, max: i32) -> i64 {
23    let sensors = parse_input(input);
24
25    for sensor in &sensors {
26        // only seek the points around the sensor range
27        let r = sensor.distance() + 1; // increase range by 1
28        let x = sensor.position.0;
29        let y = sensor.position.1;
30
31        for point in get_manhatten_border(x, y, r) {
32            if !(point.0 >= 0 && point.0 <= max && point.1 >= 0 && point.1 <= max) {
33                continue;
34            }
35            if sensors.iter().all(|s| !s.in_range_point(&point)) {
36                return point.0 as i64 * FREQUENZY_FACTOR + point.1 as i64;
37            }
38        }
39    }
40
41    panic!("Ther isn't any position");
42}

44fn get_manhatten_border(x: i32, y: i32, r: i32) -> HashSet<Point> {
45    let mut border = HashSet::new();
46    for i in 0..=r {
47        // math rotation
48        border.insert(Point(x + (r - i), y - i));
49        border.insert(Point(x + i, y + (r - i)));
50        border.insert(Point(x - i, y - (r - i)));
51        border.insert(Point(x - (r - i), y + i));
52    }
53    border
54}

56fn count_none_positions(sensors: &Vec<Sensor>, y: i32, min_x: i32, max_x: i32) -> i32 {
57    // check beacons on y
58    let beacons_on_y: HashSet<i32> = sensors
59        .iter()
60        .filter(|s| s.beacon.1 == y)
61        .map(|s| s.beacon.0)
62        .collect();
63
64    // remove beacon count
65    let mut count = -(beacons_on_y.len() as i32);
66
67    for x in min_x..=max_x {
68        for sensor in sensors {
69            if sensor.in_range(x, y) {
70                count += 1;
71                break;
72            }
73        }
74    }
75    count
76}

78#[derive(Debug, Hash, PartialEq, Eq)]
79struct Point(i32, i32);
80
81struct Sensor {
82    position: Point,
83    beacon: Point,
84}
85
86impl Sensor {
87    fn new(position: Point, beacon: Point) -> Self {
88        Self { position, beacon }
89    }
90
91    /// Returns the manhatten distance between the sensor positon and the beacon position.
92    fn distance(&self) -> i32 {
93        (self.position.0 - self.beacon.0).abs() + (self.position.1 - self.beacon.1).abs()
94    }
95
96    /// Checks if the distance between the sensor and the beacon is
97    /// smaller then the distance between the sensor and the given point.
98    fn in_range(&self, x: impl Into<i32>, y: impl Into<i32>) -> bool {
99        let squared = (self.position.0 - x.into()).abs() + (self.position.1 - y.into()).abs();
100        squared <= self.distance()
101    }
102
103    fn in_range_point(&self, point: &Point) -> bool {
104        self.in_range(point.0, point.1)
105    }
106}

108fn parse_i32(hay: Option<regex::Match<'_>>) -> i32 {
109    hay.unwrap().as_str().parse().unwrap()
110}

112fn parse_input(input: impl Into<String>) -> Vec<Sensor> {
113    let input = input.into();
114    let re = Regex::new("=(-?\\d*)").unwrap();
115    let mut result: Vec<Sensor> = Vec::with_capacity(input.lines().count());
116
117    for line in input.lines() {
118        let nums: Vec<i32> = re
119            .captures_iter(line)
120            .map(|cap| parse_i32(cap.get(1)))
121            .collect();
122
123        if nums.len() != 4 {
124            panic!(
125                "The input of day 15 has some issues - the count should be 4 but is {}\nLine:'{}",
126                nums.len(),
127                line,
128            );
129        }
130
131        let sensor = Point(nums[0], nums[1]);
132        let beacon = Point(nums[2], nums[3]);
133        result.push(Sensor::new(sensor, beacon));
134    }
135    result
136}
